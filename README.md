  };
  const selected = new Set();
  const comparison = document.getElementById('comparison');
  const compareMessage = document.getElementById('compare-message');
  const modal = document.getElementById('detail-modal');
  let previousFocus;

  document.getElementById('menu-toggle').addEventListener('click', e => {
    const menu = document.getElementById('mobile-menu');
    const open = menu.classList.toggle('hidden') === false;
    e.currentTarget.setAttribute('aria-expanded',String(open));
    e.currentTarget.setAttribute('aria-label',open ? 'Tutup menu' : 'Buka menu');
  });
  document.querySelectorAll('#mobile-menu a').forEach(link => link.addEventListener('click',() => {
    document.getElementById('mobile-menu').classList.add('hidden');
    document.getElementById('menu-toggle').setAttribute('aria-expanded','false');
  }));
  document.querySelectorAll('.filter').forEach(button => button.addEventListener('click',() => {
    document.querySelectorAll('.filter').forEach(item => item.setAttribute('aria-pressed',String(item===button)));
    document.querySelectorAll('.model-card').forEach(card => {card.hidden = button.dataset.filter !== 'all' && card.dataset.category !== button.dataset.filter;});
  }));
  function renderComparison(){
    comparison.replaceChildren();
    for(const id of selected){
      const model=models[id];
      const card=document.createElement('article');
      card.className='rounded-2xl border border-[#cbdfe3] bg-white p-5';
      const title=document.createElement('h4'); title.className='font-bold'; title.textContent=model.name;
      card.append(title);
      for(const [label,value] of [['Kategori',model.category],['Karakter penggunaan',model.use],['Kenyamanan',model.comfort],['Fitur utama',model.feature]]){
        const line=document.createElement('p');line.className='mt-3 text-sm leading-relaxed';
        const strong=document.createElement('strong');strong.textContent=label+': ';
        line.append(strong,document.createTextNode(value));card.append(line);
      }
      comparison.append(card);
    }
    compareMessage.textContent=selected.size ? `${selected.size} dari 3 pilihan dibandingkan.` : 'Belum ada pilihan. Pilih model untuk mulai membandingkan.';
  }
  document.querySelectorAll('[data-compare]').forEach(button => button.addEventListener('click',() => {
    const id=button.dataset.compare;
    if(selected.has(id)) selected.delete(id);
    else if(selected.size<3) selected.add(id);
    else {compareMessage.textContent='Maksimal tiga pilihan. Batalkan salah satu pilihan untuk menggantinya.';return;}
    button.setAttribute('aria-pressed',String(selected.has(id)));
    renderComparison();
  }));
  function closeModal(){modal.classList.add('hidden');modal.classList.remove('flex');document.body.style.overflow='';previousFocus?.focus();}
  document.querySelectorAll('[data-detail]').forEach(button => button.addEventListener('click',() => {
    const model=models[button.dataset.detail]; previousFocus=button;
    document.getElementById('modal-title').textContent=model.name;
    document.getElementById('modal-category').textContent=model.category;
    document.getElementById('modal-description').textContent=model.description;
    document.getElementById('modal-use').textContent=model.use+'. '+model.feature+'.';
    modal.classList.remove('hidden');modal.classList.add('flex');document.body.style.overflow='hidden';
    document.getElementById('close-modal').focus();
  }));
  document.getElementById('close-modal').addEventListener('click',closeModal);
  document.getElementById('modal-action').addEventListener('click',closeModal);
  modal.addEventListener('click',e=>{if(e.target===modal)closeModal()});
  document.addEventListener('keydown',e=>{
    if(e.key==='Escape' && !modal.classList.contains('hidden'))closeModal();
    if(e.key==='Tab' && !modal.classList.contains('hidden')){
      const stops=[...modal.querySelectorAll('button,a')];const first=stops[0],last=stops[stops.length-1];
      if(e.shiftKey && document.activeElement===first){e.preventDefault();last.focus()}
      else if(!e.shiftKey && document.activeElement===last){e.preventDefault();first.focus()}
    }
  });
  document.getElementById('preferences-form').addEventListener('submit',e=>{
    e.preventDefault();
    const category=document.getElementById('category').value;
    const time=document.getElementById('time').value;
    document.getElementById('form-error').classList.toggle('hidden',Boolean(category&&time));
    const result=document.getElementById('preference-result');
    result.classList.toggle('hidden',!(category&&time));
    if(category&&time)document.getElementById('summary-text').textContent=`Kategori: ${category}. Waktu yang diinginkan: ${time}.`;
  });
  renderComparison();
